<!DOCTYPE html>

<html>

<head>

  <base target="_top">

  <style>

    /* Reset margins and completely hide scrollbars */

    * { 

      margin: 0; 

      padding: 0; 

      box-sizing: border-box; 

    }

    body, html { 

      width: 100%; 

      height: 100%; 

      overflow: hidden; 

      background-color: #000; 

    }

    #signage-container {

      width: 100vw;

      height: 100vh;

      display: flex;

      justify-content: center;

      align-items: center;

      overflow: hidden;

      position: relative;

    }

    img {

      width: 100vw;

      height: 100vh;

      /* 'fill' forces the image to stretch to the 16:9 screen boundaries without cropping */

      object-fit: fill; 

      object-position: center;

      display: none;

    }

    img.active {

      display: block;

    }

    /* Centers the loading message with no alignment shifts */

    #loading {

      color: #ffffff;

      font-family: sans-serif;

      font-size: 1.5rem;

      text-align: center;

      position: absolute;

    }

  </style>

</head>

<body>


  <div id="signage-container">

    <div id="loading">Loading signage...</div>

  </div>


  <script>

    let images = [];

    let currentIndex = 0;

    const slideDuration = 10000; // Time per image in milliseconds (10 seconds)


    window.onload = function() {

      google.script.run

        .withSuccessHandler(initializeSignage)

        .withFailureHandler(showError)

        .getImageUrls();

    };


    function initializeSignage(urls) {

      const loadingDiv = document.getElementById('loading');

      if (!urls || urls.length === 0) {

        loadingDiv.innerText = 'No images found in the folder.';

        return;

      }

      

      images = urls;

      const container = document.getElementById('signage-container');

      

      // Remove the loading text element entirely so nothing lingers in the background

      if (loadingDiv) {

        loadingDiv.remove();

      }


      // Preload images into DOM

      images.forEach((url, index) => {

        const img = document.createElement('img');

        img.src = url;

        img.id = 'slide-' + index;

        if (index === 0) img.classList.add('active');

        container.appendChild(img);

      });


      // Start the loop

      setInterval(nextSlide, slideDuration);

    }


    function nextSlide() {

      if (images.length <= 1) return;


      const currentImg = document.getElementById('slide-' + currentIndex);

      if (currentImg) currentImg.classList.remove('active');


      currentIndex = (currentIndex + 1) % images.length;


      const nextImg = document.getElementById('slide-' + currentIndex);

      if (nextImg) nextImg.classList.add('active');

    }


    function showError(err) {

      document.getElementById('loading').innerText = 'Error loading signage: ' + err.message;

    }

  </script>

</body>

</html>
